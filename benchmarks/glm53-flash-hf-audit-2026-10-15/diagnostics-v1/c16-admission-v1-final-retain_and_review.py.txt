"""Retain completed C16 evidence and independently recompute all decisions on Mac."""
from __future__ import annotations
import hashlib,json,math,pathlib,shlex,subprocess,sys,tarfile
BASE=pathlib.Path(__file__).parent
ROOT='/home/kirya/local-spark/experiments/glm53-mxfp4-audit-2026-10-15/full-model-v1'
SSH=['ssh','-o','BatchMode=yes','-o','ConnectTimeout=10','spark']

def sha(p):
    with p.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()

def main():
    receipt=json.loads(subprocess.check_output([*SSH,shlex.join(['cat',ROOT+'/c16-admission-v1-evidence-check.json'])],timeout=30))
    assert receipt['status']=='complete',receipt
    assert receipt['verified_after_pack'] and receipt['restored_health']==200 and receipt['source_files']==211
    (BASE/'spark-evidence-receipt.json').write_text(json.dumps(receipt,indent=2))
    archive=BASE/'evidence.tar.gz'
    assert not archive.exists()
    with archive.open('xb') as f:
        subprocess.run([*SSH,shlex.join(['cat',receipt['archive']])],stdout=f,check=True,timeout=240)
    assert sha(archive)==receipt['archive_sha256'] and archive.stat().st_size==receipt['archive_bytes']
    root=BASE/'evidence';root.mkdir()
    with tarfile.open(archive) as tf:
        seen=set();size=0
        for member in tf.getmembers():
            p=pathlib.PurePosixPath(member.name)
            assert member.isfile() and not p.is_absolute() and '..' not in p.parts and member.name not in seen
            seen.add(member.name);size+=member.size
        assert size<4*1024**3
        tf.extractall(root,filter='data')
    verifier=root/'verify.py'
    assert sha(verifier)==json.loads((BASE/'source-freeze.json').read_bytes())['operations_sha256']['verify.py']
    out=BASE/'mac-independent-review.json'
    subprocess.run([sys.executable,'-B',str(verifier),'--root',str(root),'--output',str(out)],check=True,timeout=900)
    spark=json.loads((root/'spark-independent-review.json').read_bytes());mac=json.loads(out.read_bytes())
    largest=[0.0]
    def compare(a,b):
        assert type(a)==type(b),(type(a),type(b))
        if isinstance(a,dict):
            assert a.keys()==b.keys()
            for k in a:compare(a[k],b[k])
        elif isinstance(a,list):
            assert len(a)==len(b)
            for x,y in zip(a,b,strict=True):compare(x,y)
        elif isinstance(a,float):
            assert math.isfinite(a) and math.isfinite(b)
            largest[0]=max(largest[0],abs(a-b))
            assert math.isclose(a,b,rel_tol=1e-10,abs_tol=1e-12),(a,b)
        else:assert a==b,(a,b)
    compare(spark,mac)
    result=mac.get('recomputed_review',{})
    proof={'status':'passed','archive_sha256':sha(archive),'archive_bytes':archive.stat().st_size,
        'payloads':mac['payloads'],'source_files':mac['source_files'],
        'measurement_complete':mac['measurement_complete'],'all_decisions_identical':True,
        'maximum_float64_difference':largest[0],'CPU_absolute_tolerance':1e-12,'CPU_relative_tolerance':1e-10,
        'model_KL_limit':1e-5,'model_absolute_NLL_limit':1e-4,'quality_qualified':False,
        'original_service_restored':True,'native_mhc_unchanged':True,'input_FP32_enabled':False,
        'compression_numerical_gate_passed':result.get('compression_numerical_gate_passed'),
        'all_quality_groupings_matched':result.get('all_quality_groupings_matched'),
        'C16_matched_numerical_gate_passed':result.get('C16_matched_numerical_gate_passed'),
        'policy_gate_passed':result.get('policy_gate_passed'),
        'compression_speed_gate_by_window':result.get('compression_speed_gate_by_window'),
        'all_timed_generated_ids_match':result.get('all_timed_generated_ids_match'),
        'timed_token_sequence_pairs':result.get('timed_token_sequence_pairs'),
        'dispatch_startup_preparation_passed':result.get('dispatch_startup_preparation_passed'),
        'dispatch_cache_unchanged_after_startup':result.get('dispatch_cache_unchanged_through_model_work'),
        'exact16_dispatch_mechanism_passed':mac.get('exact16_dispatch_mechanism_passed'),
        'exact16_decode_ranges':len(mac.get('decode_dispatch_mechanism',[])),
        'worker_profiles':len(mac['profiles'])}
    (BASE/'cross-host-verification.json').write_text(json.dumps(proof,indent=2)+'\n')
    if result:
        compact={k:v for k,v in result.items() if k not in ('numerical_comparisons','CPU_memory_observations','startup_resource_observations')}
        compact['numerical_comparisons']=[{k:v for k,v in row.items() if k!='steps'} for row in result['numerical_comparisons']]
        compact['cross_host_verification']=proof
        (BASE/'compact-review.json').write_text(json.dumps(compact,indent=2)+'\n')
    print(json.dumps(proof))

if __name__=='__main__':main()

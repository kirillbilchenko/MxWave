# Pinned FlashInfer's FC1 activation consumes [linear/up, gate]. Marlin consumes
# [gate, up]. Permute whole rows and their scale rows; preserve every weight bit.
NATIVE_FC1_PROJECTIONS = ('up_proj', 'gate_proj')

def sha(path):
 with pathlib.Path(path).open('rb')as f:return hashlib.file_digest(f,'sha256').hexdigest()
def save(path,value):
 path=pathlib.Path(path);temporary=path.with_suffix('.incomplete');temporary.write_text(json.dumps(value,indent=2,allow_nan=False)+'\n');temporary.replace(path)
def telemetry():
 p=subprocess.run(['nvidia-smi','--query-gpu=clocks.sm,clocks.mem,temperature.gpu,power.draw,pstate','--format=csv,noheader,nounits'],capture_output=True,text=True)
 return dict(returncode=p.returncode,values=p.stdout.strip(),epoch=time.time())
def nbytes(t):return t.numel()*t.element_size()
def metrics(a,b):
 a,b=a.float(),b.float();diff=a-b;finite=bool(torch.isfinite(a).all()and torch.isfinite(b).all())
 return dict(finite=finite,
  changed_values=int((a!=b).sum()),values=a.numel(),max_abs=float(diff.abs().max())if finite else None,
  relative_l2=float(torch.linalg.vector_norm(diff)/torch.linalg.vector_norm(b).clamp_min(1e-30))if finite else None,
  rms_error=float(diff.square().mean().sqrt())if finite else None,reference_rms=float(b.square().mean().sqrt())if finite else None)
def retain_tensor(path,tensor):
 cpu=tensor.detach().cpu().contiguous();raw=cpu.reshape(-1).view(torch.uint8).numpy().tobytes();path.write_bytes(raw)
 return dict(path=path.name,shape=list(cpu.shape),dtype=str(cpu.dtype),bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest())

def load_native(fixture,manifest,rank):
 """Load source FP4 bytes unchanged and reindex only the exact E8M0 scales."""
 records={(r['expert'],r['projection'],r['suffix']):r for r in manifest['tensors']}
 def source(e,p,s):
  record=records[e,p,s];path=fixture/record['path'];assert sha(path)==record['sha256']
  t=torch.frombuffer(bytearray(path.read_bytes()),dtype=torch.uint8).reshape(record['shape'])
  lo,hi=rank*1024,(rank+1)*1024
  if p=='down_proj':
   stride=32 if s=='weight_scale'else 2;t=t[:,lo//stride:hi//stride]
  else:t=t[lo:hi]
  return t.contiguous()
 weights,scales=[],[]
 for names,N,K in [(NATIVE_FC1_PROJECTIONS,2048,4096),(('down_proj',),4096,1024)]:
  weight=torch.empty(288,N,K//2,dtype=torch.uint8,device='cuda')
  scale=torch.empty(288,K//128,N*4,dtype=torch.uint8,device='cuda')
  for e in range(288):
   raw=torch.cat([source(e,p,'weight_packed')for p in names]);weight[e].copy_(raw)
   sf=torch.cat([source(e,p,'weight_scale')for p in names]);assert not (sf==255).any()
   # Each output row's four adjacent UE8M0 bytes become a K128-major packed word.
   view=sf.reshape(N,K//128,4).permute(1,0,2).contiguous().reshape(K//128,N*4)
   assert torch.equal(view.reshape(K//128,N,4).permute(1,0,2).reshape_as(sf),sf)
   scale[e].copy_(view)
  weights.append(weight);scales.append(scale)
 return weights,scales

parts=[f"**Live C16 admission status: {stamp}.** Phase `{state['phase']}`; service paused: {state['service_paused']}.\n"]

    if values['summaries']:
        parts.append('Complete timing cells below are preliminary and await independent archive verification; every declared repetition is included.\n\n| Arm | Case | Window | Trials | Median tok/s | Median TTFT |\n|---|---|---:|---:|---:|---:|\n'+''.join(f"| {row['arm']} | {row['case']} | {row['window_ms']} ms | {row['trials']} | {row['median_TPS']:.3f} | {row['median_TTFT_seconds']:.6f} s |\n" for row in values['summaries']))

    audit=json.loads((BASE/'fp32-kl-audit.json').read_bytes())

The qualified route already uses FP32 accumulation; split-K=2 changes its grouping. Reverting to the qualified route is the tested numerical recovery and preserves compression.     paired=json.loads((BASE/'paired-numerical-progress.json').read_bytes())
    parts.append(f"Completed controller review (independent archive review pending): numerical gate {paired['compression_numerical_gate_passed']}, policy gate {paired['policy_gate_passed']}, all {paired['timed_token_sequence_pairs']} timed output pairs match: {paired['all_timed_generated_ids_match']}. A separate collector resumes byte-checked retention and copies head traces. Results, 211 frozen sources, verifier and gates are unchanged; this is archival recovery only.\n')
block='<!-- C16_LIVE_START -->\n'+'\n'.join(parts)+'<!-- C16_LIVE_END -->\n\n'
records=[]
for name in ('GLM53_HANDOFF.md','GLM53_CURRENT_STATE.md','GLM53_C16_ADMISSION.md'):
    path=WORK/'docs'/name;old=path.read_text()
    if '<!-- C16_LIVE_START -->' in old:
        start=old.index('<!-- C16_LIVE_START -->');end=old.index('<!-- C16_LIVE_END -->',start)+len('<!-- C16_LIVE_END -->\n\n')

        heading,tail=old.split('\n\n',1);new=heading+'\n\n'+block+tail
    path.write_text(new)
    records.append({'document':str(path),'sha256':hashlib.sha256(path.read_bytes()).hexdigest()})

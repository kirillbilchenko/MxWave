"""Start the prepared bounded component only after verified budget-run restoration."""
from __future__ import annotations
import json
import pathlib
import shlex
import subprocess
import time

base=pathlib.Path(__file__).parent
root='/home/kirya/local-spark/experiments/glm53-mxfp4-audit-2026-10-15/full-model-v1'
assert json.loads((base/'actual-preparation.json').read_bytes())['status']=='prepared'
assert not (base/'actual-launch.json').exists()
program=r'''
import json,pathlib,subprocess,sys,time
r=pathlib.Path(sys.argv[1]);sys.path.insert(0,str(r/'full-prefillbudgetv3-src'));import full_model_control as c
state=json.loads((r/'quality-model-controller-prefillbudgetv3.json').read_bytes())
receipt=json.loads((r/'prefill-budget-v3-evidence-check.json').read_bytes())
assert state['phase']=='complete'and not state['service_paused']and receipt['status']=='complete'
assert receipt['restored_health']==200 and receipt['verified_after_pack']
assert c.health(8000)
prepared=json.loads((r/'moe-native-actual-v3-prepared.json').read_bytes())
assert [c.identity(c.inspect(i,c.ORIGINAL_IDS[i]))for i in (0,1)]==prepared['originals']
assert not (r/'moe-native-actual-v3-controller.json').exists()
checked=json.loads(c.run(0,['python3','-S','-B',str(r/'moe-native-actual-v3-src/run_native_actual.py'),'--check-only'],timeout=180))
proc=subprocess.Popen(['python3','-S','-B','-u',str(r/'moe-native-actual-v3-src/run_native_actual.py')],
 stdout=(r/'moe-native-actual-v3-controller.log').open('xb'),stderr=subprocess.STDOUT,
 stdin=subprocess.DEVNULL,start_new_session=True)
result=dict(status='launched',pid=proc.pid,epoch=time.time(),preflight=checked,
 previous_model_evidence_sha256=receipt['archive_sha256'],independent_restore_watchdog_required=True)
c.write(r/'moe-native-actual-v3-launch.json',result);print(json.dumps(result))
'''
raw=subprocess.check_output(['ssh','-o','BatchMode=yes','spark',shlex.join(['python3','-S','-B','-c',program,root])],timeout=240)
(base/'actual-launch.json').write_bytes(raw)
awake=subprocess.Popen(['/usr/bin/caffeinate','-di','-t','3600'],stdin=subprocess.DEVNULL,stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,start_new_session=True)
identity=subprocess.check_output(['ps','-p',str(awake.pid),'-o','pid=,lstart=,command='],text=True).strip()
(base/'actual-awake-lease.json').write_text(json.dumps(dict(pid=awake.pid,identity=identity,started_epoch=time.time()),indent=2)+'\n')
print(raw.decode())

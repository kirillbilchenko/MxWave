"""Recompute memory diagnostic evidence without a GPU or model imports."""
from __future__ import annotations
import argparse,csv,hashlib,json,pathlib,sys

def sha(p):
 with p.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()

def counters(path,labels,metrics):
 rows=list(csv.reader(path.read_text().splitlines()))
 header=next((row for row in rows if row and row[0]=='ID'),None)
 if header is None:return {'complete':False,'error':'No raw NCU table','kernels':[]}
 start=rows.index(header);units=dict(zip(header,rows[start+1],strict=True))
 kernels=[dict(zip(header,row,strict=True)) for row in rows[start+2:] if len(row)==len(header) and row[0].isdigit()]
 found=set();result=[]
 for row in kernels:
  name=row['Kernel Name'];matches=[x for x in labels if name.startswith(x+'/')]
  label=max(matches,key=len) if matches else None
  if label:found.add(label)
  values={};missing=[]
  for metric in metrics:
   raw=row.get(metric,'');value=None
   try:value=float(raw.replace(',',''))
   except ValueError:missing.append(metric)
   values[metric]={'value':value,'unit':units.get(metric),'raw':raw}
  result.append({'id':row['ID'],'name':name,'label':label,'metrics':values,'unavailable':missing})
 required=('gpu__time_duration.sum','lts__t_bytes.sum','smsp__inst_executed.sum')
 complete=found==set(labels) and bool(result) and all(x['label'] is not None and all(x['metrics'][m]['value'] is not None for m in required) for x in result)
 return {'complete':complete,'expected_labels':labels,'observed_labels':sorted(found),'kernels':result,
         'DRAM_byte_counters_available':False,'counter_replay_is_API_latency':False}

def main():
 p=argparse.ArgumentParser();p.add_argument('--root',type=pathlib.Path,required=True);p.add_argument('--output',type=pathlib.Path,required=True);a=p.parse_args();r=a.root
 inv=json.loads((r/'inventory.json').read_bytes())['files'];assert len({x['path'] for x in inv})==len(inv)
 for row in inv:
  rel=pathlib.PurePosixPath(row['path']);assert not rel.is_absolute() and '..' not in rel.parts
  f=r/rel;assert f.is_file() and not f.is_symlink() and f.stat().st_size==row['bytes'] and sha(f)==row['sha256'],row['path']
 prepared=json.loads((r/'prepared.json').read_bytes());plan=json.loads((r/'plan.json').read_bytes())
 actual={str(f.relative_to(r/'source')):sha(f) for f in (r/'source').rglob('*') if f.is_file() and f.suffix in ('.py','.toml')}
 assert actual==prepared['bindings'][0]['code']==prepared['bindings'][1]['code'] and len(actual)==plan['expected_source_files']
 assert all(actual[n]==v for n,v in plan['parent_source_sha256'].items())
 assert all(actual[n]==v for n,v in plan['source_sha256'].items())
 sys.path.insert(0,str(r/'source'));from kda_memory_probe import assess,protocol,operation_names,METRICS
 assert prepared['protocol']==plan['protocol']==protocol()
 state=json.loads((r/'controller.json').read_bytes());post=json.loads((r/'post-restoration-check.json').read_bytes())
 assert state['phase'] in ('complete','failed_restored') and not state['service_paused']
 assert post['health']==200 and post['originals']==prepared['originals'] and post['controller_terminal'] and post['watchdog_terminal']
 assert state['restoration']['status']=='complete' and state['restoration']['health']==200
 checks=[];profiles=[]
 for entry in state['runs']:
  rank,stage=entry['rank'],entry['stage'];path=r/f'rank{rank}-{stage}-report.json'
  if not path.exists():checks.append({'rank':rank,'stage':stage,'diagnostic_complete':False,'error':'report absent'});continue
  report=json.loads(path.read_bytes());assert report==json.loads((r/f'rank{rank}/{stage}.json').read_bytes())
  assert report['source_sha256']==actual['benchmark_kda_memory.py'] and report['protocol']==protocol()
  if report['status']!='complete':checks.append({'rank':rank,'stage':stage,'diagnostic_complete':False,'error':report.get('error')});continue
  result=assess(report);assert result==report['assessment']
  assert {k:v for k,v in result.items() if k!='summaries'}==entry['independent_assessment']
  checks.append({'rank':rank,'stage':stage,'exit_code':entry['exit_code'],'OOMKilled':entry['OOMKilled'],**result,'CUDA_peak_bytes':report['CUDA_peak_bytes']})
  if stage=='profile':profiles.append({'rank':rank,**counters(r/f'rank{rank}/ncu.csv',['kda_memory/'+n for n in operation_names()],METRICS)})
 unchanged=state.get('source_after')==prepared['bindings'] and state.get('input_bindings_after')==prepared['input_bindings']
 complete=len(checks)==4 and unchanged and all(x['diagnostic_complete'] and not x.get('exit_code') and not x.get('OOMKilled') for x in checks) and len(profiles)==2 and all(x['complete'] for x in profiles)
 out={'status':'complete','diagnostic_complete':complete,'payloads':len(inv),'source_files':len(actual),'processes':checks,'profiles':profiles,'source_and_fixture_bindings_unchanged':unchanged,'original_service_restored':True,'model_quality_or_serving_speed_qualified':False,'model_CPU_saving_demonstrated':False,'limits':['Same layer-zero/rank-zero weights on both physical GPUs; representative kernel diagnostic, not all layers.','Logical payload, measured L2 requests and measured LPDDR traffic are distinct.','NCU cold-replay timings do not replace ordinary serving measurements.','Repeat-control failures remain included; no optimization is qualified by this diagnostic.']}
 a.output.write_text(json.dumps(out,indent=2,allow_nan=False)+'\n');print(json.dumps({k:v for k,v in out.items() if k not in ('processes','profiles','limits')}))

if __name__=='__main__':main()

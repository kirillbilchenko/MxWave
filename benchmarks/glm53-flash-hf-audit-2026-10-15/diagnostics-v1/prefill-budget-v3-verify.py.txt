def sha(p):
 with p.open("rb")as f:return hashlib.file_digest(f,"sha256").hexdigest()
def close(a,b):
 if isinstance(a,dict):
  assert a.keys()==b.keys(),(a.keys(),b.keys())
  for k in a:close(a[k],b[k])
 elif isinstance(a,list):
  assert len(a)==len(b)
  for x,y in zip(a,b,strict=True):close(x,y)
 elif isinstance(a,float):assert math.isclose(a,b,rel_tol=1e-10,abs_tol=1e-12),(a,b)
 else:assert a==b,(a,b)
def trace_review(path,groups,rank):
    with gzip.open(path,"rt") if path.name.endswith(".gz") else path.open() as handle:
        trace=json.load(handle)
    events=trace["traceEvents"]
    tag=re.compile(r"^mxwave_c8_(.+)_step_(\d+)$")
    ranges=[]
    for e in events:
        m=tag.match(e.get("name",""))
        if m and e.get("cat")=="user_annotation":
            ranges.append({"label":m[1],"sequence":int(m[2]),"pid":e["pid"],"tid":e["tid"],
             "start":e["ts"],"end":e["ts"]+e["dur"],"cuda_graph_launches":0,
             "kernels":[]})
    ranges.sort(key=lambda r:r["start"])
    runtime={}
    for e in events:
        if e.get("cat") not in ("cuda_runtime","cuda_driver"):continue
        key=e.get("args",{}).get("correlation")
        if key is None:continue
        assert key not in runtime
        owners=[i for i,r in enumerate(ranges)if (r["pid"],r["tid"])==(e["pid"],e["tid"])
          and r["start"]<=e["ts"]<=r["end"]]
        assert len(owners)<=1
        owner=owners[0] if owners else None
        runtime[key]=owner
        if owner is not None and e.get("name","").startswith("cudaGraphLaunch"):
            ranges[owner]["cuda_graph_launches"]+=1
    unassigned=0
    total=0
    for e in events:
        if e.get("cat")!="kernel":continue
        total+=1
        owner=runtime.get(e.get("args",{}).get("correlation"))
        if owner is None:
            unassigned+=1;continue
        ranges[owner]["kernels"].append(e)
    rows=[]
    for r in ranges:
        group=groups[r["label"]]
        workers={w["rank"]:w for w in group["worker_trace"]}
        event=workers[rank]["events"][r["sequence"]]
        assert event["sequence"]==r["sequence"]
        variants=Counter()
        durations=defaultdict(float)
        order=[]
        for e in sorted(r["kernels"],key=lambda e:e["ts"]):
            args=e.get("args",{})
            value={"name":e["name"],"grid":args.get("grid"),"block":args.get("block"),
                   "shared_memory_bytes":args.get("shared memory")}
            key=json.dumps(value,sort_keys=True,separators=(",",":"))
            variants[key]+=1;durations[e["name"]]+=float(e.get("dur",0))
            order.append(key)
        variant_rows=[{**json.loads(k),"count":v}for k,v in sorted(variants.items())]
        digest=lambda v:hashlib.sha256(json.dumps(v,sort_keys=True,separators=(",",":")).encode()).hexdigest()
        rows.append({"label":r["label"],"sequence":r["sequence"],
          "batch":event["prepared"],"total_scheduled_tokens":event["total_scheduled_tokens"],
          "cuda_graph_launches":r["cuda_graph_launches"],"kernel_events":len(r["kernels"]),
          "kernel_variant_multiset_sha256":digest(variant_rows),
          "observed_kernel_order_sha256":digest(order),
          "kernel_variants":variant_rows,
          "kernel_duration_sum_us":sum(durations.values()),
          "top_kernel_duration_sums_us":[{"name":n,"duration_us":v}for n,v in sorted(durations.items(),key=lambda t:(-t[1],t[0]))[:12]]})
    assert sum(x["kernel_events"] for x in rows)+unassigned==total
    labels=sorted({r["label"]for r in rows})
    coverage={label:{"profiled_steps":sum(r["label"]==label for r in rows),
       "recorded_steps":len(next(w for w in groups[label]["worker_trace"]if w["rank"]==rank)["events"])}
       for label in labels}
    return {"path":str(path),"rank":rank,"labels":labels,"steps":rows,
      "kernel_events":total,"unassigned_kernel_events":unassigned,"coverage":coverage,
      "ownership":"Device correlation -> host launch inside mxwave_c8 label/step annotation; same PID/TID",
      "limits":["Kernel variants or order do not prove numerical root cause.",
       "Duration sums are not wall time or measured memory bandwidth.",
       "Sampling/logprob work outside the model annotation remains explicitly unassigned.",
       "Profiler active window may omit later forwards."]}
def verify_decode_mechanism(profiles):
 rows=[]
 for profile in profiles:
  arm="candidate"if profile["labels"][0].startswith("candidate-")else"control"
  assert len(profile["labels"])==1
  expected=8 if "-c8-" in profile["labels"][0] else 16
  selected=[s for s in profile["steps"]if not s["batch"]["has_prefill"]and s["batch"]["num_tokens_after_padding"]==expected]
  assert selected,("No expected decode shape observed",expected,profile["path"])
  for step in selected:
   variants=step["kernel_variants"]
   rebuild=sum(v["count"]for v in variants if "bf16triplebitmap_decompress_kernel"in v["name"].lower())
   fused=[v for v in variants if "BF16TripleBitmap_MM_Kernel"in v["name"]]
   calls=sum(v["count"]for v in fused)
   assert rebuild==0 and step["cuda_graph_launches"]>0
   if arm=="candidate":
    assert calls==68
    assert sum(v["count"]for v in fused if v["grid"][1]==197)==34
    assert sum(v["count"]for v in fused if v["grid"][1]==64)==34
   else:assert calls==0
   rows.append({"arm":arm,"rank":profile["rank"],"label":step["label"],"sequence":step["sequence"],
    "actual_rows":step["batch"]["num_tokens"],"physical_rows":expected,"graph_launches":step["cuda_graph_launches"],
    "reconstruction_calls":rebuild,"compressed_KDA_calls":calls})
 assert {(r["arm"],r["rank"],r["physical_rows"])for r in rows}=={(a,r,m)for a in("control","candidate")for r in(0,1)for m in(16,)}
 return rows

"""Check pinned call signatures and fixture-adapter cleanup without importing CUDA."""
from __future__ import annotations
import ast
import hashlib
import inspect
import json
import pathlib
import types

base=pathlib.Path(__file__).parent
baseline=pathlib.Path('/private/tmp/mxwave-prefill-budget-v3/cpu-source')
installed=pathlib.Path('/private/tmp/mxwave-moe-backend-v1/installed-source')
source=base/'benchmark_native_experts.py'
tree=ast.parse(source.read_text())
adapter=next(n for n in tree.body if isinstance(n,ast.FunctionDef)and n.name=='load_current_fixture')
bound={}

def signature(path,name):
    node=next(n for n in ast.walk(ast.parse(path.read_text()))if isinstance(n,ast.FunctionDef)and n.name==name)
    bound[str(path)]=hashlib.sha256(path.read_bytes()).hexdigest()
    args=node.args
    parameters=[]
    combined=[*args.posonlyargs,*args.args]
    defaults=[inspect.Parameter.empty]*(len(combined)-len(args.defaults))+[None]*len(args.defaults)
    for i,(arg,default)in enumerate(zip(combined,defaults)):
        kind=inspect.Parameter.POSITIONAL_ONLY if i<len(args.posonlyargs)else inspect.Parameter.POSITIONAL_OR_KEYWORD
        parameters.append(inspect.Parameter(arg.arg,kind,default=default))
    if args.vararg:parameters.append(inspect.Parameter(args.vararg.arg,inspect.Parameter.VAR_POSITIONAL))
    for arg,default in zip(args.kwonlyargs,args.kw_defaults):
        parameters.append(inspect.Parameter(arg.arg,inspect.Parameter.KEYWORD_ONLY,default=inspect.Parameter.empty if default is None else None))
    if args.kwarg:parameters.append(inspect.Parameter(args.kwarg.arg,inspect.Parameter.VAR_KEYWORD))
    return inspect.Signature(parameters)

loader_signature=signature(baseline/'benchmark_moe.py','load_fixture')
repack_signature=signature(installed/'vllm/_custom_ops.py','gptq_marlin_repack')
fused_signature=signature(installed/'vllm/model_executor/layers/fused_moe/experts/marlin_moe.py','fused_marlin_moe')
for node in ast.walk(tree):
    if isinstance(node,ast.Call)and isinstance(node.func,ast.Attribute)and node.func.attr=='fused_marlin_moe':
        assert all(k.arg is not None for k in node.keywords)
        fused_signature.bind(*[object()for _ in node.args],**{k.arg:object()for k in node.keywords})

sentinel=object();empty=object();weight=types.SimpleNamespace(device='cuda:test');calls=[]
def original(*args):
    repack_signature.bind(*args)
    assert args==(weight,empty,4096,2048,4,False)
    calls.append(args)
    return sentinel
ops=types.SimpleNamespace(gptq_marlin_repack=original)
def make_empty(n,*,dtype,device):
    assert n==0 and dtype=='int32'and device==weight.device
    return empty
namespace={'torch':types.SimpleNamespace(empty=make_empty,int32='int32')}
exec(compile(ast.Module(body=[adapter],type_ignores=[]),str(source),'exec'),namespace)
run=namespace['load_current_fixture']
def loader(*args,**kwargs):
    loader_signature.bind(*args,**kwargs)
    assert args==('fixture',{'layer':23},2,1,'cuda:test')
    return ops.gptq_marlin_repack(weight,4096,2048,4,False)
assert run(loader,ops,'fixture',{'layer':23},1,'cuda:test')is sentinel
assert ops.gptq_marlin_repack is original and len(calls)==1
def failed_loader(*args):
    loader_signature.bind(*args)
    raise ValueError('intentional fixture failure')
try:run(failed_loader,ops,'fixture',{'layer':23},1,'cuda:test')
except ValueError:pass
else:raise AssertionError('Fixture error was suppressed')
assert ops.gptq_marlin_repack is original
try:loader_signature.bind('fixture',{},2,1,'cuda:test',current_api=True)
except TypeError:pass
else:raise AssertionError('The failed v3 call was not rejected')
result=dict(status='passed',CUDA_imported=False,source_sha256=hashlib.sha256(source.read_bytes()).hexdigest(),
    checked_source_sha256=bound,old_loader_call_rejected=True,explicit_empty_permutation_checked=True,
    adapter_restored_after_success_and_failure=True,fused_call_keywords_match_pinned_signature=True)
(base/'actual-api-CPU-check.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result))

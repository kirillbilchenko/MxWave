"""Independently inspect serialized scales, untouched bytes and sampled FP8 rounding."""
from __future__ import annotations
import hashlib, json, math, struct
from pathlib import Path
import numpy as np

ROOT = Path(__file__).resolve().parent

def tensor_file(path):
    with path.open('rb') as stream:
        length=struct.unpack('<Q',stream.read(8))[0]
        header=json.loads(stream.read(length))
    return header,np.memmap(path,mode='r',dtype=np.uint8,offset=8+length)

source,source_data=tensor_file(ROOT/'modal-checkpoint/model.safetensors')
output,output_data=tensor_file(ROOT/'modal-fp8-checkpoint/model.safetensors')
review=json.loads((ROOT/'modal-fp8-checkpoint/conversion-review.json').read_bytes())
codes=np.arange(127,dtype=np.uint8)
exponent=(codes>>3).astype(np.int32)
mantissa=(codes&7).astype(np.float32)
levels=np.where(exponent==0,mantissa*np.float32(2**-9),
                np.ldexp(np.float32(1)+mantissa/np.float32(8),exponent-7)).astype(np.float32)
midpoints=(levels[:-1]+levels[1:])*np.float32(.5)
rng=np.random.default_rng(42)
rows=[];checked=0;copied=0
for row in review['tensors']:
    name=row['name'];a=source[name];b=output[name]
    original=source_data[slice(*a['data_offsets'])]
    encoded=output_data[slice(*b['data_offsets'])]
    if row['operation']=='bitwise_copy':
        assert a['shape']==b['shape'] and a['dtype']==b['dtype']
        assert np.array_equal(original,encoded),name
        assert hashlib.sha256(encoded).hexdigest()==row['sha256']
        copied+=1
        continue
    assert a['dtype']=='BF16' and b['dtype']=='F8_E4M3'
    assert not np.any((encoded&127)==127)
    n,k=a['shape'];bf16=original.view('<u2').reshape(n,k)
    fp8=encoded.reshape(n,k)
    scale_desc=output[name.removesuffix('.weight')+'.weight_scale_inv']
    assert scale_desc['dtype']=='F32' and scale_desc['shape']==[n//128,k//128]
    scales=output_data[slice(*scale_desc['data_offsets'])].view('<f4').reshape(n//128,k//128)
    assert np.isfinite(scales).all() and (scales>0).all()
    blocks=n//128*(k//128)
    indices=sorted({0,blocks-1,*map(int,rng.choice(blocks,size=8,replace=False))})
    for block in indices:
        i,j=divmod(block,k//128);selection=np.s_[i*128:(i+1)*128,j*128:(j+1)*128]
        values=(bf16[selection].astype(np.uint32)<<16).view(np.float32)
        maximum=np.max(np.abs(values))
        scale=np.float32(1) if maximum==0 else np.float32(maximum/np.float32(448))
        assert scales[i,j]==scale,(name,block,'scale')
        normalized=np.clip(values/scale,np.float32(-448),np.float32(448))
        absolute=np.abs(normalized)
        nearest=np.searchsorted(midpoints,absolute,side='left')
        midpoint=midpoints[np.minimum(nearest,len(midpoints)-1)]
        nearest=nearest+((absolute==midpoint)&((nearest&1)==1))
        expected=nearest.astype(np.uint8)|(np.signbit(normalized).astype(np.uint8)<<7)
        assert np.array_equal(fp8[selection],expected),(name,block,'rounding')
        checked+=values.size
    rows.append({'name':name,'sampled_blocks':len(indices),'different_codes':0})
assert copied==54 and len(rows)==42
result={'status':'verified','copied_tensors_bitwise_equal':copied,
        'quantized_tensors_finite_and_scales_positive':len(rows),
        'numpy_independent_sampled_fp8_codes':checked,'different_codes':0,
        'full_fp8_rounding_audit':False,'sampled_tensors':rows,
        'script_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'training_performed':False,'native_reload_or_acceptance_tested':False}
(ROOT/'fp8-numpy-review.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='sampled_tensors'}))

"""Review passive paired target/drafter captures using retained CPU tensors."""

from __future__ import annotations

import hashlib
import json
from pathlib import Path
from typing import Any

import numpy as np
from safetensors.numpy import load_file


def prefix_matches(actual: list[int], expected: list[int]) -> int:
    """Count matching consecutive tokens, stopping at the first disagreement."""
    return next((i for i, pair in enumerate(zip(actual, expected, strict=True))
                 if pair[0] != pair[1]), len(actual))


def values(tensor: Any, dtype: str) -> Any:
    """Decode retained BF16 bits while preserving all other numeric dtypes."""
    if dtype == "torch.bfloat16":
        return (tensor.view(np.uint16).astype(np.uint32) << 16).view(np.float32)
    return tensor


def difference(a: Any, b: Any) -> dict[str, Any]:
    """Summarize an aligned tensor pair without treating hidden drift as task loss."""
    assert a.shape == b.shape and np.isfinite(a).all() and np.isfinite(b).all()
    x, y = a.astype(np.float64).ravel(), b.astype(np.float64).ravel()
    delta = x - y
    norm = float(np.linalg.norm(x) * np.linalg.norm(y))
    return {
        "shape": list(a.shape), "elements": int(a.size),
        "unequal_elements": int(np.count_nonzero(a != b)),
        "max_abs": float(np.abs(delta).max()),
        "rmse": float(np.sqrt(np.mean(delta * delta))),
        "relative_l2_to_modelopt": float(np.linalg.norm(delta) / max(np.linalg.norm(y), 1e-30)),
        "cosine": float(np.dot(x, y) / norm) if norm else None,
    }


def path(ids: Any, scores: Any) -> list[int]:
    """Select a conditional greedy path from retained candidate scores."""
    previous = 0
    selected = []
    for step in range(7):
        previous = int(scores[step, previous].argmax())
        selected.append(int(ids[step, previous]))
    return selected


def proposal_review(data: dict[str, Any], expected: list[int]) -> dict[str, Any]:
    """Locate proposal errors and screen selector strength on fixed captured states."""
    ids = data["draft_ids"].reshape(7, 16)
    unary = data["draft_values"].reshape(7, 16)
    edges = data["draft_scores"].reshape(7, 16, 16)
    native = data["draft_proposal"].reshape(7).tolist()
    assert path(ids, edges) == native
    previous = 0
    rows = []
    for step, token in enumerate(native):
        chosen = int(np.flatnonzero(ids[step] == token)[0])
        hits = np.flatnonzero(ids[step] == expected[step])
        wanted = int(hits[0]) if len(hits) else None
        scores = edges[step, previous]
        rows.append({
            "position": step + 1, "proposed": token, "expected": expected[step],
            "expected_candidate_rank": wanted + 1 if wanted is not None else None,
            "chosen_candidate_rank": chosen + 1,
            "selected_minus_expected_score": (
                float(scores[chosen] - scores[wanted]) if wanted is not None else None
            ),
        })
        previous = chosen
    sensitivity = {}
    for factor in (0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5):
        candidate_scores = (
            edges if factor == 1 else unary[:, None, :] + factor * (edges - unary[:, None, :])
        )
        proposed = path(ids, candidate_scores)
        sensitivity[str(factor)] = {
            "proposed": proposed, "known_common_prefix_matches": prefix_matches(proposed, expected),
        }
    return {
        "proposed": native, "expected_common_history": expected,
        "known_common_prefix_matches": prefix_matches(native, expected),
        "positions": rows, "selector_strength_sensitivity": sensitivity,
        "sensitivity_scope": "Offline fixed-state hypothesis screen; no serving or quality claim",
    }


def counters(text: str) -> dict[str, float]:
    """Extract labelled speculative totals without merging unlike counters."""
    result = {}
    for line in text.splitlines():
        if line.startswith("vllm:spec_decode_"):
            key, value = line.rsplit(" ", 1)
            if key.split("{", 1)[0].endswith("_total"):
                result[key] = float(value)
    return result


def analyze(root: Path) -> dict[str, Any]:
    """Check request controls, runtime identities and the first three matched anchors."""
    inputs = json.loads((root / "inputs.json").read_bytes())["cases"][0]
    prompt = inputs["original_case"]["prompt_tokens"]
    output = inputs["histories"]["ct_mxfp8"]["token_ids"]
    assert output[:32] == inputs["histories"]["modelopt"]["token_ids"][:32]
    report: dict[str, Any] = {
        "schema": "mxwave-drafter-context-review-v1", "timing_claim": False,
        "source_sha256": {}, "controls": {}, "ranks": {},
    }
    frames: dict[str, Any] = {}
    weights: dict[str, Any] = {}
    for arm in ("ct", "modelopt"):
        measurement = json.loads((root / f"{arm}-measurement.json").read_bytes())
        assert measurement["status"] == "complete"
        rows = measurement["cases"]
        assert len(rows) == 3
        changes = []
        patterns = []
        for row in rows:
            assert row["result"]["token_ids"] == output[:32]
            before, after = counters(row["metrics_before"]), counters(row["metrics_after"])
            changes.append({key: after[key] - value for key, value in before.items()})
            patterns.append([a["tokens"] for a in row["result"]["arrivals"]])
        report["controls"][arm] = {
            "three_outputs_match_retained_prefix": True,
            "speculative_counters_repeat": changes[0] == changes[1] == changes[2],
            "stream_chunk_patterns_repeat": patterns[0] == patterns[1] == patterns[2],
            "counter_deltas": changes, "chunk_patterns": patterns,
        }
        frames[arm], weights[arm] = {}, {}
        for rank in (0, 1):
            directory = root / f"{arm}-r{rank}"
            weights[arm][rank] = json.loads(
                (directory / f"draft-runtime-weights-r{rank}.json").read_bytes()
            )
            loaded = []
            for meta_path in sorted(directory.glob("context-code_32-*.json")):
                meta = json.loads(meta_path.read_bytes())
                payload = meta_path.with_suffix(".safetensors")
                assert hashlib.sha256(payload.read_bytes()).hexdigest() == meta["payload_sha256"]
                data = load_file(payload)
                assert meta["index"] == len(loaded) and meta["draft_graph"] == [1, 8]
                assert np.array_equal(data["draft_proposal"], data["returned_proposal"])
                assert data["request_temperature"].tolist() == [0.0]
                assert path(data["draft_ids"].reshape(7, 16),
                            data["draft_scores"].reshape(7, 16, 16)) == (
                    data["draft_proposal"].reshape(7).tolist()
                )
                if "target_logits" in data:
                    logits = values(data["target_logits"], meta["tensor_dtypes"]["target_logits"])
                    chosen = logits.argmax(axis=-1).tolist()
                    sampled = int(data["target_num_sampled"].reshape(-1)[0])
                    actual = data["target_sampled_tokens"].reshape(-1)[:sampled].tolist()
                    meta["raw_target_argmax_matches_native_sampled_prefix"] = (
                        chosen[:sampled] == actual
                    )
                loaded.append((meta, data))
            assert len(loaded) >= 4
            frames[arm][rank] = loaded
    for rank in (0, 1):
        a, b = weights["ct"][rank], weights["modelopt"][rank]
        names = sorted(a.keys() | b.keys())
        mismatches = [name for name in names if a.get(name) != b.get(name)]
        comparisons = []
        for index, expected_offset in enumerate((1, 4, 5)):
            ma, da = frames["ct"][rank][index]
            mb, db = frames["modelopt"][rank][index]
            starts = [int(d["sample_positions"].reshape(-1)[0]) - prompt for d in (da, db)]
            assert starts == [expected_offset, expected_offset]
            valid_a = ma["target_tokens"] - int(da["target_num_rejected"].reshape(-1)[0])
            valid_b = mb["target_tokens"] - int(db["target_num_rejected"].reshape(-1)[0])
            assert valid_a == valid_b and ma["target_tokens"] == mb["target_tokens"]
            for name in ("target_positions", "target_input_ids"):
                assert np.array_equal(da[name][:valid_a], db[name][:valid_b])
            assert np.array_equal(da["draft_query_ids"], db["draft_query_ids"])
            assert np.array_equal(da["draft_query_positions"], db["draft_query_positions"])
            stats = {}
            for name in [*(f"aux_layer_{x}" for x in (5, 14, 24, 33, 42)),
                         "combined_target_hidden", "draft_hidden"]:
                x = values(da[name], ma["tensor_dtypes"][name])
                y = values(db[name], mb["tensor_dtypes"][name])
                if name != "draft_hidden":
                    x, y = x[:valid_a], y[:valid_b]
                stats[name] = difference(x, y)
            expected = output[expected_offset:expected_offset + 7]
            next_a = frames["ct"][rank][index + 1][1]
            next_b = frames["modelopt"][rank][index + 1][1]
            comparisons.append({
                "generated_prefix_tokens": expected_offset,
                "valid_context_rows": valid_a, "input_histories_and_shapes_aligned": True,
                "feature_differences": stats,
                "ct": proposal_review(da, expected),
                "modelopt": proposal_review(db, expected),
                "next_verification_accepted": {
                    "ct": int(next_a["target_num_sampled"].reshape(-1)[0]) - 1,
                    "modelopt": int(next_b["target_num_sampled"].reshape(-1)[0]) - 1,
                },
            })
        report["ranks"][str(rank)] = {
            "runtime_weight_tensors": len(names),
            "runtime_weight_mismatches": mismatches,
            "captured_frames": {arm: len(frames[arm][rank]) for arm in frames},
            "raw_argmax_sample_checks": {
                arm: [meta["raw_target_argmax_matches_native_sampled_prefix"]
                      for meta, _ in frames[arm][rank]
                      if "raw_target_argmax_matches_native_sampled_prefix" in meta]
                for arm in frames
            },
            "matched_anchors": comparisons,
        }
    report["controls_pass"] = all(
        row["speculative_counters_repeat"] and row["stream_chunk_patterns_repeat"]
        for row in report["controls"].values()
    )
    for source in ("inputs.json", "plan.json", "ct-measurement.json", "modelopt-measurement.json"):
        report["source_sha256"][source] = hashlib.sha256((root / source).read_bytes()).hexdigest()
    report["source_sha256"]["analyzer"] = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    return report


if __name__ == "__main__":
    import sys

    result = analyze(Path(sys.argv[1]))
    output_path = Path(sys.argv[2])
    assert not output_path.exists()
    output_path.write_text(json.dumps(result, indent=2) + "\n")
    print(json.dumps({
        "controls_pass": result["controls_pass"],
        "runtime_weight_mismatches": {
            rank: row["runtime_weight_mismatches"] for rank, row in result["ranks"].items()
        },
        "anchors": [
            {"offset": row["generated_prefix_tokens"],
             "accepted": row["next_verification_accepted"],
             "ct_proposed": row["ct"]["proposed"],
             "modelopt_proposed": row["modelopt"]["proposed"]}
            for row in result["ranks"]["0"]["matched_anchors"]
        ],
    }))

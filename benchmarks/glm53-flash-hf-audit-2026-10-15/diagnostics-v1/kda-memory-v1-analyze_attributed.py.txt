"""Bind offline NVTX exports to original counters and summarize the prefill diagnostic."""
from __future__ import annotations
import argparse,csv,hashlib,importlib.util,json,pathlib,statistics,sys

def sha(p):
 with p.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()

def table(path):
 rows=list(csv.reader(path.read_text().splitlines()));h=next(x for x in rows if x and x[0]=='ID');i=rows.index(h)
 units=dict(zip(h,rows[i+1],strict=True));data={row[0]:dict(zip(h,row,strict=True)) for row in rows[i+2:] if len(row)==len(h) and row[0].isdigit()}
 return units,data

def main():
 p=argparse.ArgumentParser();p.add_argument('--root',type=pathlib.Path,required=True);p.add_argument('--output',type=pathlib.Path,required=True);a=p.parse_args();r=a.root
 inv={x['path']:x for x in json.loads((r/'inventory.json').read_bytes())['files']}
 def checked(relative):
  path=r/relative;row=inv[relative];assert path.stat().st_size==row['bytes'] and sha(path)==row['sha256'];return path
 prepared=json.loads(checked('prepared.json').read_bytes())
 for name,digest in prepared['bindings'][0]['code'].items():assert sha(checked('source/'+name))==digest
 sys.path.insert(0,str(r/'source'));from kda_memory_probe import assess,operation_names,METRICS
 verifier=checked('verify.py');assert sha(verifier)=='73a93dba15717a0a84cc2f07b92a1e43cf9023baea7c811ba7bb129ab5ad0f3d'
 spec=importlib.util.spec_from_file_location('frozen_memory_verifier',verifier);module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module)
 original=json.loads((r/'spark-independent-review.json').read_bytes());assert original['status']=='complete' and original['original_service_restored'] and original['source_and_fixture_bindings_unchanged']
 assert len(original['processes'])==4 and all(x['diagnostic_complete'] and x['exit_code']==0 and not x['OOMKilled'] for x in original['processes'])
 profiles=[];timings=[];copies=[];decoders=[];coupled=[];bindings=[]
 for rank in (0,1):
  raw=checked(f'rank{rank}/ncu.csv');named=checked(f'rank{rank}/ncu-nvtx.csv');binary=checked(f'rank{rank}/kda-memory.ncu-rep');receipt=checked(f'rank{rank}/nvtx-export.json');export=json.loads(receipt.read_bytes())
  assert export['binary_report_sha256']==sha(binary) and export['export_sha256']==sha(named)
  assert export['profiler_sha256']==prepared['input_bindings'][rank]['ncu_sha256'] and not export['GPU_workload_run']
  units,old=table(raw);named_units,new=table(named);assert units==named_units and old.keys()==new.keys()
  for key in old:
   assert {k:v for k,v in old[key].items() if k!='Kernel Name'}=={k:v for k,v in new[key].items() if k!='Kernel Name'}
   assert new[key]['Kernel Name'].endswith('/'+old[key]['Kernel Name'])
  profile=module.counters(named,['kda_memory/'+x for x in operation_names()],METRICS);assert profile['complete'] and len(profile['kernels'])==28
  profiles.append({'rank':rank,**profile})
  bindings.append({'rank':rank,'original_CSV_sha256':sha(raw),'named_CSV_sha256':sha(named),'binary_sha256':sha(binary),'export_receipt_sha256':sha(receipt),'all_original_counter_values_and_units_identical':True,'only_kernel_names_changed':True})
  report=json.loads(checked(f'rank{rank}-timing-report.json').read_bytes());review=assess(report);assert review==report['assessment']
  timings.append({'rank':rank,**review})
  summaries={(x['name'],x['mode'],x['cache']):x for x in review['summaries']}
  for mode in ('eager','graph'):
   for cache in ('warm','pressure'):
    for size in (4,128):
     candidates=[summaries[f'copy-{size}MiB-b{block}',mode,cache] for block in (1024,4096,16384)]
     fastest=max(candidates,key=lambda x:x['logical_payload_GB_per_second'])
     copies.append({'rank':rank,'MiB':size,'mode':mode,'cache':cache,'maximum_observed_logical_GBs':fastest['logical_payload_GB_per_second'],'block_from':fastest['name'],'that_repeat_control_passed':fastest['repeat_control_passed'],'all_three_controls_passed':all(x['repeat_control_passed'] for x in candidates),'all_configurations_retained':True,'calibrated_hardware_peak_claimed':False})
    for fixture in ('kda-input-l0-r0','kda-output-l0-r0'):
     d=summaries[fixture+'/decode',mode,cache];c=summaries[fixture+'/copy-dense',mode,cache]
     decoders.append({'rank':rank,'fixture':fixture,'mode':mode,'cache':cache,'decoder_us':d['median_us'],'copy_us':c['median_us'],'decoder_logical_GBs':d['logical_payload_GB_per_second'],'copy_logical_GBs':c['logical_payload_GB_per_second'],'decoder_to_copy_payload_rate':d['logical_payload_GB_per_second']/c['logical_payload_GB_per_second'],'decoder_repeat_control_passed':d['repeat_control_passed'],'copy_repeat_control_passed':c['repeat_control_passed']})
     for rows in (54,256,2048):
      dense=summaries[f'{fixture}/M{rows}/dense',mode,cache];combined=summaries[f'{fixture}/M{rows}/decode-dense',mode,cache]
      coupled.append({'rank':rank,'fixture':fixture,'rows':rows,'mode':mode,'cache':cache,'dense_us':dense['median_us'],'decode_dense_us':combined['median_us'],'descriptive_extra_us':combined['median_us']-dense['median_us'],'dense_control_passed':dense['repeat_control_passed'],'combined_control_passed':combined['repeat_control_passed'],'paired_speed_qualification_claimed':False})
 report={'status':'complete','source_sha256':sha(pathlib.Path(__file__)),'original_inventory_sha256':sha(r/'inventory.json'),'original_verifier_diagnostic_complete':original['diagnostic_complete'],'offline_attribution_complete':True,'all_22_labels_per_GPU_present':True,'all_28_kernel_records_per_GPU_present':True,'bindings':bindings,'profiles':profiles,'timings':timings,'copy_baselines':copies,'decoder_comparisons':decoders,'coupled_comparisons':coupled,'failed_repeat_controls_by_rank':[len(x['unstable_repeat_controls']) for x in timings],'original_service_restored':True,'GPU_experiment_run':False,'model_policy_or_measurement_changed':False,'automatic_promotion_allowed':False,'limits':['Original raw CSV omitted NVTX names; the original attribution-failure verdict is retained.','Offline exports change names only; all kernel IDs, units and counter values are identical to the original CSV.','Every 2% repeat-control failure stays in the result; no reruns or discarded trials.','Payload-rate comparisons are descriptive across separate operation measurements, not paired optimization speed gates.','L2 request bytes are measured; LPDDR bytes and a hardware bandwidth ceiling remain unmeasured.','Both devices run the same two rank-zero/layer-zero fixtures; this is not complete model coverage.','Prefill dense/coupled operation differences do not establish whole-model API effects.']}
 a.output.write_text(json.dumps(report,indent=2,allow_nan=False)+'\n')
 print(json.dumps({k:v for k,v in report.items() if k not in ['profiles','timings','bindings','copy_baselines','decoder_comparisons','coupled_comparisons','limits']}))

if __name__=='__main__':main()

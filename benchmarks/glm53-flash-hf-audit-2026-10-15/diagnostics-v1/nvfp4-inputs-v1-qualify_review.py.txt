"""Summarize the existing mean-KL gate without replacing it with maximum KL."""
from __future__ import annotations
import argparse
import hashlib
import json
import statistics
from pathlib import Path

parser = argparse.ArgumentParser()
parser.add_argument('evidence', type=Path)
parser.add_argument('output', type=Path)
args = parser.parse_args()
source = args.evidence / 'repeat-review.json'
review = json.loads(source.read_bytes())
rows = []
for comparison in review['controls']:
    prefill = [r['kl'] for r in comparison['rows'] if r['position'] == 0 and r['kl'] is not None]
    decode = [r['kl'] for r in comparison['rows'] if r['position'] > 0 and r['kl'] is not None]
    conditioning = all(r['same_conditioning'] for r in comparison['rows'])
    means = [statistics.mean(prefill), statistics.mean(decode)]
    passed = comparison['layouts_match'] and conditioning and all(v < 1e-5 for v in means)
    rows.append({k: comparison[k] for k in ('left', 'right', 'layouts_match', 'generated_tokens_match')} | {
        'conditioning_matches': conditioning, 'prefill_rows': len(prefill), 'decode_rows': len(decode),
        'mean_prefill_kl': means[0], 'mean_decode_kl': means[1],
        'maximum_prefill_kl': max(prefill), 'maximum_decode_kl': max(decode),
        'gate_passed': passed})
result = {
    'review_sha256': hashlib.sha256(source.read_bytes()).hexdigest(),
    'definition': 'Mean prefill and mean decode KL each < 1e-5; matching layouts and conditioning.',
    'maximum_KL_is_diagnostic_only': True,
    'rows': rows,
    'reference_repeat_set_qualified': all(row['gate_passed'] for row in rows),
    'compressed_comparison_performed': False,
    'serving_speed_measured': False,
}
args.output.write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps(result, indent=2))

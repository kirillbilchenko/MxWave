"""Launch the authorized final admission experiment and retain exact operational identities."""
from __future__ import annotations
import hashlib,json,pathlib,shlex,subprocess,time
BASE=pathlib.Path(__file__).parent
ROOT='/home/kirya/local-spark/experiments/glm53-mxfp4-audit-2026-10-15/full-model-v1'
SSH=['ssh','-o','BatchMode=yes','-o','ConnectTimeout=10','spark']
prepared=json.loads((BASE/'preparation.json').read_bytes())
assert prepared['status']=='passed' and prepared['source_files']==217
assert not(BASE/'launch.json').exists()
awake=subprocess.Popen(['/usr/bin/caffeinate','-i','-t','7200'],stdin=subprocess.DEVNULL,
    stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL,start_new_session=True)
identity=subprocess.check_output(['ps','-p',str(awake.pid),'-o','pid=,lstart=,command='],text=True).strip()
(BASE/'awake-lease.json').write_text(json.dumps({'pid':awake.pid,'identity':identity,
    'started_epoch':time.time(),'duration_seconds':7200,'purpose':'C16 admission run and independent retention'},indent=2))
result=subprocess.run([*SSH,shlex.join(['python3','-B','-u',ROOT+'/launch-prefill-budget-v3.py'])],capture_output=True,text=True)
(BASE/'launch.log').write_text(result.stdout+result.stderr)
assert result.returncode==0,result.stdout+result.stderr
launch=json.loads(result.stdout.strip().splitlines()[-1])
(BASE/'launch.json').write_text(json.dumps(launch,indent=2))
raw=subprocess.check_output([*SSH,shlex.join(['cat',ROOT+'/prefill-budget-v3-prepared.json'])])
assert hashlib.sha256(raw).hexdigest()==prepared['prepared_sha256']
(BASE/'prepared.json').write_bytes(raw)
print(json.dumps(launch))
